# Cursor Bot Dashboard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a GitHub Pages dashboard that shows Cursor Bot adoption across 17 Humand squads, sprint by sprint, with data pulled from Jira via a scheduled GitHub Actions workflow.

**Architecture:** A Node 20 script (`fetch-data.js`) runs in GitHub Actions every 6h, calls Jira REST API using a secret token, and writes a static `data.json` to the repo. A single-file `index.html` reads `data.json` at load time and renders the dashboard with Chart.js via CDN. No build step, no backend, no CORS issues (browser never touches Jira).

**Tech Stack:** Node 20 (native `fetch`), GitHub Actions, vanilla HTML/CSS/JS, Chart.js 4.4 (CDN), Inter font (Google Fonts CDN), Jira Cloud REST API v3 + Agile API v1.

|---|---|
| Dashboard muestra "Sin datos todavía" | El workflow aún no corrió. Andá a Actions y triggerealo manualmente |
| Workflow falla con `Jira 401` | El token expiró o está mal codificado. Regenerá y actualizá el secret `JIRA_TOKEN` |
| Workflow falla con `Jira 403` | Tu usuario de Jira no tiene permisos para leer algún squad. Revisá acceso |

```javascript
// fetch-data.js — corre en GitHub Actions, genera data.json
// Requiere: Node 20+ y la env var JIRA_TOKEN (base64 de email:api_token)

const BASE_URL    = 'https://humand.atlassian.net';
const TOKEN       = process.env.JIRA_TOKEN;
const BOT_ACCOUNT = '712020:98b3a270-fe83-4788-9d35-e5b5611a7a64';
const START_DATE  = '2025-02-24';
const SQUADS = [
  'SQZB', 'SQSQ', 'SQSH', 'SQRN', 'SQRC', 'SQPM', 'SQPD',
  'SQOW', 'SQOT', 'SQKA', 'SQJG', 'SQGZ', 'SQEG', 'SQDP',
  'SQXS', 'SQCY', 'SQWH'
];

if (!TOKEN) {
  console.error('ERROR: falta la env var JIRA_TOKEN');
  process.exit(1);
}

// ── Helpers ───────────────────────────────────────────────────────────

async function jiraFetch(path, retryOnce = true) {
  const res = await fetch(`${BASE_URL}${path}`, {
    headers: {
      'Authorization': `Basic ${TOKEN}`,
      'Accept': 'application/json'
    }
  });
  if (!res.ok) {
    if (retryOnce && res.status >= 500) {
      await new Promise(r => setTimeout(r, 1000));
      return jiraFetch(path, false);
    }
    const body = await res.text().catch(() => '');
    throw new Error(`Jira ${res.status} → ${path.split('?')[0]}${body ? ` :: ${body.slice(0, 200)}` : ''}`);
  }
  return res.json();
}

async function getBoardId(squad) {
  const data = await jiraFetch(
    `/rest/agile/1.0/board?projectKeyOrId=${squad}&type=scrum&maxResults=1`
  );
  if (!data.values || data.values.length === 0) return null;
  return data.values[0].id;
}

async function getSprints(boardId) {
  const data = await jiraFetch(
    `/rest/agile/1.0/board/${boardId}/sprint?state=closed,active&maxResults=100`
  );
  const cutoff = new Date(START_DATE);
  return (data.values || [])
    .filter(s => s.startDate && new Date(s.startDate) >= cutoff)
    .sort((a, b) => new Date(a.startDate) - new Date(b.startDate))
    .map(s => ({
      id: s.id,
      name: s.name,
      startDate: s.startDate,
      endDate: s.endDate || null
    }));
}

async function getIssueCount(squad, sprintId, botOnly) {
  const assigneeClause = botOnly ? ` AND assignee = "${BOT_ACCOUNT}"` : '';
  const jql = `project = "${squad}" AND issuetype in (Subtask, "Sub-task", "Dev Task") AND status = Done AND sprint = ${sprintId}${assigneeClause}`;
  const data = await jiraFetch(
    `/rest/api/3/search?jql=${encodeURIComponent(jql)}&maxResults=0&fields=id`
  );
  return data.total || 0;
}

async function loadSquad(squad) {
  try {
    const boardId = await getBoardId(squad);
    if (!boardId) return { sprints: [], error: 'Sin board Scrum' };

    const sprints = await getSprints(boardId);
    if (sprints.length === 0) return { sprints: [], error: 'Sin sprints recientes' };

    const sprintData = [];
    for (const s of sprints) {
      const [botCount, totalCount] = await Promise.all([
        getIssueCount(squad, s.id, true),
        getIssueCount(squad, s.id, false)
      ]);
      sprintData.push({ ...s, botCount, totalCount });
    }

    return { sprints: sprintData, error: null };
  } catch (err) {
    return { sprints: [], error: err.message };
  }
}

// ── Main ──────────────────────────────────────────────────────────────

(async () => {
  console.log(`Fetching data for ${SQUADS.length} squads...`);

  const squadResults = {};
  for (const squad of SQUADS) {
    const result = await loadSquad(squad);
    const botTotal = result.sprints.reduce((sum, s) => sum + s.botCount, 0);
    const suffix = result.error ? ` (${result.error})` : ` → ${result.sprints.length} sprints, ${botTotal} bot tasks`;
    console.log(`  ${result.error ? '⚠' : '✓'} ${squad}${suffix}`);
    squadResults[squad] = result;
  }

  const output = {
    lastUpdated: new Date().toISOString(),
    squads: squadResults
  };

  const fs = await import('fs');
  fs.writeFileSync('data.json', JSON.stringify(output, null, 2));
  console.log(`\n✅ data.json escrito (${fs.statSync('data.json').size} bytes)`);
})();
```

## Task 3: GitHub Actions workflow

**Files:**
- Create: `.github/workflows/fetch-data.yml`

- [ ] **Step 1: Crear el workflow**

```yaml
name: Actualizar datos de Jira

on:
  schedule:
    - cron: '0 */6 * * *'       # cada 6 horas
  workflow_dispatch:             # disparo manual desde la UI

permissions:
  contents: write

jobs:
  fetch:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'

      - name: Fetch datos de Jira
        env:
          JIRA_TOKEN: ${{ secrets.JIRA_TOKEN }}
        run: node fetch-data.js

      - name: Commit y push data.json (solo si cambió)
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          git add data.json
          if git diff --staged --quiet; then
            echo "Sin cambios en data.json"
          else
            git commit -m "chore: actualizar data.json [skip ci]"
            git push
          fi
```

// Stats de un squad: totales y % adopción global del squad
function computeSquadStats(squadData) {
  const sprints = squadData.sprints || [];
  const totalBot = sprints.reduce((s, x) => s + x.botCount, 0);
  const totalAll = sprints.reduce((s, x) => s + x.totalCount, 0);
  const pct = totalAll > 0 ? (totalBot / totalAll) * 100 : 0;
  return { totalBot, totalAll, pct, sprintCount: sprints.length };
}

// % adopción de un squad en un sprint específico
function sprintPct(sprint) {
  return sprint.totalCount > 0 ? (sprint.botCount / sprint.totalCount) * 100 : 0;
}

// Lista única de sprints cronológicos (por startDate) entre todos los squads
function allSprintsChrono(rawData) {
  const map = new Map();
  for (const squad of SQUADS) {
    const sprints = rawData.squads[squad]?.sprints || [];
    for (const s of sprints) {
      const key = s.startDate;
      if (!map.has(key)) {
        map.set(key, { startDate: s.startDate, endDate: s.endDate, label: null });
      }
    }
  }
  const arr = Array.from(map.values())
    .sort((a, b) => new Date(a.startDate) - new Date(b.startDate));
  arr.forEach((s, i) => { s.label = `S${i + 1}`; });
  return arr;
}

// Adopción promedio del sprint (promedio simple entre squads con actividad ese sprint)
function avgAdoptionForSprint(rawData, startDate) {
  const pcts = [];
  for (const squad of SQUADS) {
    const sprint = (rawData.squads[squad]?.sprints || []).find(s => s.startDate === startDate);
    if (sprint && sprint.totalCount > 0) pcts.push(sprintPct(sprint));
  }
  if (pcts.length === 0) return null;
  return pcts.reduce((a, b) => a + b, 0) / pcts.length;
}

function formatPct(n) {
  return n == null ? '--' : `${n.toFixed(0)}%`;
}

function pctColorClass(pct) {
  if (pct == null) return '';
  if (pct >= 75) return 'high';
  if (pct >= 50) return 'med';
  return 'low';
}

function pctColor(pct) {
  if (pct >= 75) return '#10b981';
  if (pct >= 50) return '#2563eb';
  return '#f59e0b';
}

function renderSidebarSquads() {
  const container = document.getElementById('sidebar-squads');
  const withStats = SQUADS.map(squad => ({
    squad,
    stats: state.squadStats[squad],
    hasData: (state.rawData.squads[squad]?.sprints || []).length > 0
  }));

  // Ordenar: primero los que tienen datos (por % desc), después los sin datos
  withStats.sort((a, b) => {
    if (a.hasData && !b.hasData) return -1;
    if (!a.hasData && b.hasData) return 1;
    return (b.stats?.pct || 0) - (a.stats?.pct || 0);
  });

  container.innerHTML = withStats.map(({ squad, stats, hasData }) => {
    const pct = hasData ? formatPct(stats.pct) : '--';
    const cls = hasData ? '' : 'no-data';
    return `
      <div class="sidebar-item ${cls}" data-view="${squad}">
        <span>${squad}</span>
        <span class="pct">${pct}</span>
      </div>
    `;
  }).join('');

  // Attach click listeners
  container.querySelectorAll('.sidebar-item').forEach(el => {
    el.addEventListener('click', () => switchView(el.dataset.view));
  });
  document.querySelector('[data-view="global"]').addEventListener('click', () => switchView('global'));
}

function renderTopBottom(sprintsAll, squadsWithData) {
  const lastSprint = sprintsAll[sprintsAll.length - 1];
  if (!lastSprint) return;

  const pairs = squadsWithData
    .map(squad => {
      const sprint = (state.rawData.squads[squad]?.sprints || []).find(s => s.startDate === lastSprint.startDate);
      if (!sprint || sprint.totalCount === 0) return null;
      return { squad, pct: sprintPct(sprint) };
    })
    .filter(Boolean);

  const top = [...pairs].sort((a, b) => b.pct - a.pct).slice(0, 3);
  const bottom = [...pairs].sort((a, b) => a.pct - b.pct).slice(0, 3);

  function renderRows(container, rows) {
    document.getElementById(container).innerHTML = rows.length === 0
      ? `<div style="color:var(--text-muted);font-size:13px;padding:12px 0">Sin datos en el último sprint</div>`

          <span class="rank-pct ${pctColorClass(r.pct)}">${formatPct(r.pct)}</span>
        </div>
      `).join('');
  }

  renderRows('top-rows', top);
  renderRows('bottom-rows', bottom);
}

function switchView(view) {
  state.currentView = view;
  document.querySelectorAll('.sidebar-item').forEach(el => {
    el.classList.toggle('active', el.dataset.view === view);
  });
  if (view === 'global') {
    document.getElementById('view-global').classList.remove('hidden');
    document.getElementById('view-squad').classList.add('hidden');
    renderGlobalView();
  } else {
    document.getElementById('view-global').classList.add('hidden');
    document.getElementById('view-squad').classList.remove('hidden');
    renderSquadView(view);
  }
}

  Object.values(state.charts).forEach(c => c?.destroy?.());

async function boot() {
  try {
    // Intentar data.json primero, fallback a data.sample.json para dev local
    let res = await fetch('./data.json');
    if (!res.ok) {
      console.warn('data.json no disponible, usando data.sample.json');
      res = await fetch('./data.sample.json');
      if (!res.ok) throw new Error('No hay data.json ni data.sample.json. El workflow de GitHub Actions aún no corrió.');
    }

    state.rawData = await res.json();

    // Computar stats de cada squad
    for (const squad of SQUADS) {
      const sq = state.rawData.squads[squad];
      if (sq) state.squadStats[squad] = computeSquadStats(sq);
    }

    renderSidebarSquads();
    renderTimestamp();
    renderGlobalView();

    document.getElementById('loading').classList.add('hidden');
  } catch (err) {
    document.getElementById('loading-text').textContent = 'Sin datos todavía';
    document.getElementById('loading-sub').textContent = err.message;
    document.querySelector('#loading .spinner').style.display = 'none';
    console.error(err);
  }
}

boot();
